# Host component checks. The playable native host/renderer are built separately
# by packaging/macos/native and gpu; this harness does not start a game.
cmake_minimum_required(VERSION 3.24)
# All host test executables target Apple Silicon.
set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Host component target architecture")
project(gow3_host_components LANGUAGES C)
if(NOT APPLE)
    message(FATAL_ERROR "This harness is for macOS host component validation.")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
enable_testing()
find_package(Threads REQUIRED)
find_package(SDL3 CONFIG REQUIRED)
find_path(FFMPEG_INCLUDE_DIR libavcodec/avcodec.h REQUIRED)
foreach(component avcodec avutil swscale)
    find_library(FFMPEG_${component}_LIBRARY NAMES ${component} REQUIRED)
    list(APPEND FFMPEG_LIBRARIES "${FFMPEG_${component}_LIBRARY}")
endforeach()
if(NOT EXISTS "${ROOT}/third_party/LibAtrac9/C/src/libatrac9.h")
    message(FATAL_ERROR "Run: git submodule update --init third_party/LibAtrac9")
endif()
file(GLOB ATRAC9_SOURCES CONFIGURE_DEPENDS "${ROOT}/third_party/LibAtrac9/C/src/*.c")
add_library(atrac9 STATIC ${ATRAC9_SOURCES})
set_target_properties(atrac9 PROPERTIES C_STANDARD 99)

function(host_test name)
    add_executable(${name} ${ARGN})
    target_include_directories(${name} PRIVATE "${ROOT}" "${ROOT}/src" "${FFMPEG_INCLUDE_DIR}")
    target_compile_definitions(${name} PRIVATE GOW3_HOST_COMPONENT_TEST=1)
    # Existing tests use assert for both calls and checks, even in Release builds.
    target_compile_options(${name} PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    target_link_libraries(${name} PRIVATE Threads::Threads)
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY "${ROOT}" TIMEOUT 30)
endfunction()

host_test(content-test "${ROOT}/tests/test_content.c" "${ROOT}/src/runtime_content.c")
host_test(services-test "${ROOT}/tests/test_services.c")
host_test(pad-map-test "${ROOT}/tests/test_pad_map.c")
host_test(pad-test "${ROOT}/tests/test_pad.c")
host_test(audio-downmix-test "${ROOT}/tests/test_audio_downmix.c")
host_test(audio-host-test "${ROOT}/tests/test_audio_host.c")
foreach(test pad-map-test pad-test audio-downmix-test audio-host-test)
    target_link_libraries(${test} PRIVATE SDL3::SDL3)
endforeach()
host_test(mp3-test "${ROOT}/tests/test_mp3.c" "${ROOT}/src/runtime_mp3.c")
target_link_libraries(mp3-test PRIVATE atrac9 ${FFMPEG_LIBRARIES})
host_test(videodec-test "${ROOT}/tests/test_videodec.c")
target_link_libraries(videodec-test PRIVATE ${FFMPEG_LIBRARIES})

option(GOW3_HOST_VULKAN "Build standalone Vulkan tools (not the game renderer)" ON)
option(GOW3_TEST_VULKAN "Run the Vulkan submission test; requires a working ICD" OFF)
if(GOW3_HOST_VULKAN)
    find_package(Vulkan REQUIRED)
    add_executable(gow3-gpu-capabilities "${ROOT}/tools/gpu_capabilities.c")
    target_include_directories(gow3-gpu-capabilities PRIVATE "${ROOT}/gpu/shadps4")
    add_executable(vulkan-host-smoke "${ROOT}/tests/vulkan_host_main.c" "${ROOT}/src/vulkan_smoke.c")
    foreach(tool gow3-gpu-capabilities vulkan-host-smoke)
        target_compile_options(${tool} PRIVATE -Wall -Wextra -Werror)
        target_link_libraries(${tool} PRIVATE Vulkan::Vulkan)
    endforeach()
    if(GOW3_TEST_VULKAN)
        add_test(NAME vulkan-host-smoke COMMAND vulkan-host-smoke)
        set_tests_properties(vulkan-host-smoke PROPERTIES TIMEOUT 30)
    endif()
elseif(GOW3_TEST_VULKAN)
    message(FATAL_ERROR "GOW3_TEST_VULKAN requires GOW3_HOST_VULKAN")
endif()
message(STATUS "Host tests only: no PS4 code execution, game renderer or playable game")
